### §2 现状事实清单（已核对源码，供你了解上下文，不需要重复回答）

技术栈：Electron 33 + 原生 JS（无前端框架），主窗口 renderer 有 Node 集成（可直接 `require`），`<webview>` 跑 9 个网页版 AI。

1. **协议内核已有一个**：`adapters/api-adapter.js`（397 行）实现了 OpenAI `/chat/completions` + SSE 流解析：
   - 请求体硬编码 `{ model, messages:[{role:'user',content}], stream:true }`（L159-165）——**单条消息，无历史**；
   - SSE 手动解析，`delta.content || delta.reasoning_content` 都会累积（L219-224）；
   - 附件内联：图片转 base64 data URL、文本类直接拼进 prompt，单文件上限 4MB，PDF/docx 明确报 `UNSUPPORTED_FORMAT_FOR_API`（L39-51, L100-142）；
   - 错误分类 AUTH_FAILED / RATE_LIMITED / SERVER_ERROR（L184-195）；
   - 超时用 AbortController（L167-168）。
2. **配置只读、来源单一**：`adapters/dsh-config.js`（329 行）从 DSH 客户端的 `~/.dsh/settings.yaml` + `.credentials.yaml` 读提供方（L219-293）；内置端点表只有 4 家（deepseek / agne / google / openai，L194-199）；模型列表来自 yaml 的 `models` 字段或硬编码 DEFAULT_MODELS（L251-253）——**没有 /v1/models 拉取**。这个目录属于另一个客户端程序，meeting-lab 只是"蹭读"。
3. **席位机制**：shell/index.html L534-545，`API_CANDIDATES` = 每个有密钥的提供方取**第一个模型**生成一个候选席位（当前实测 agne/google/deepseek 共 3 席）；灯箱点亮才入会；API 席位格子是只读输出区（`.txt`），**没有输入框、没有会话历史**。
4. **发送通路已接**：人工分发甲板可对 API 席位发文本（`rawAdapter.sendText`，L1352-1353）。
5. **归档已有**：网页版 AI 的回复已自动归档到 `replies/<席位名>.md`（一个窗口一个文件、追加式）；API 席位的回复尚未接归档。
6. **当前实测可用提供方**：agne（12 模型）、google（gemini-3.6-flash 等）、deepseek（chat/reasoner），密钥都在本地凭据文件里。

async function* streamChat() {
  const controller = new AbortController();

  const timeout = setTimeout(() => controller.abort(), 120000);

  try {
    const response = await fetch(
      `${baseURL.replace(/\/$/, "")}/chat/completions`,
      {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${apiKey}`,
          "Content-Type": "application/json"
        },
        body: JSON.stringify({
          model,
          messages: [
            {
              role: "user",
              content: "请用三句话介绍一下通辽会议。"
            }
          ],
          stream: true,
          stream_options: {
            include_usage: true
          }
        }),
        signal: controller.signal
      }
    );

    if (!response.ok) {
      const body = await response.text();
      throw new Error(`HTTP ${response.status}: ${body}`);
    }

    const reader = response.body.getReader();
    const decoder = new TextDecoder();

    let buffer = "";

    while (true) {
      const { done, value } = await reader.read();

      if (done) break;

      buffer += decoder.decode(value, {
        stream: true
      });

      const lines = buffer.split("\n");
      buffer = lines.pop() || "";

      for (const rawLine of lines) {
        const line = rawLine.trim();

        if (!line.startsWith("data:")) continue;

        const payload = line.slice(5).trim();

        if (payload === "[DONE]") {
          return;
        }

        const json = JSON.parse(payload);

        if (json.usage) {
          yield {
            type: "usage",
            usage: {
              inputTokens: json.usage.prompt_tokens,
              outputTokens: json.usage.completion_tokens,
              totalTokens: json.usage.total_tokens,
              reasoningTokens:
                json.usage.completion_tokens_details
                  ?.reasoning_tokens
            }
          };
        }

        const delta = json.choices?.[0]?.delta;

        if (!delta) continue;

        if (delta.content) {
          yield {
            type: "text-delta",
            text: delta.content
          };
        }

        if (delta.reasoning_content) {
          yield {
            type: "reasoning-delta",
            text: delta.reasoning_content
          };
        }
      }
    }
  } finally {
    clearTimeout(timeout);
  }
}

const { app, safeStorage } = require("electron");
const fs = require("node:fs");
const path = require("node:path");

class SecretStore {
  constructor() {
    this.file = path.join(
      app.getPath("userData"),
      "credentials.enc.json"
    );
  }

  ensureReady() {
    if (!safeStorage.isEncryptionAvailable()) {
      throw new Error(
        "CREDENTIAL_STORAGE_UNAVAILABLE"
      );
    }
  }

  load() {
    if (!fs.existsSync(this.file)) {
      return {
        version: 1,
        credentials: {}
      };
    }

    return JSON.parse(
      fs.readFileSync(this.file, "utf8")
    );
  }

  save(data) {
    const tmp = `${this.file}.tmp`;

    fs.writeFileSync(
      tmp,
      JSON.stringify(data, null, 2),
      "utf8"
    );

    fs.renameSync(tmp, this.file);
  }

  set(id, secret) {
    this.ensureReady();

    const data = this.load();

    const encrypted =
      safeStorage.encryptString(secret);

    data.credentials[id] = {
      type: "api-key",
      cipher: "electron-safe-storage",
      data: encrypted.toString("base64"),
      updatedAt: new Date().toISOString()
    };

    this.save(data);
  }

  get(id) {
    this.ensureReady();

    const data = this.load();
    const record = data.credentials[id];

    if (!record) return null;

    const buffer = Buffer.from(
      record.data,
      "base64"
    );

    return safeStorage.decryptString(buffer);
  }
}

class RenderScheduler {
  constructor() {
    this.dirty = new Map();
    this.running = false;
  }

  mark(seatId, renderFn) {
    this.dirty.set(seatId, renderFn);

    if (!this.running) {
      this.running = true;
      requestAnimationFrame(() => this.flush());
    }
  }

  flush() {
    const jobs = [...this.dirty.values()];
    this.dirty.clear();

    const start = performance.now();

    for (const render of jobs) {
      render();

      // 避免一帧把 renderer 卡死
      if (performance.now() - start > 10) {
        break;
      }
    }

    if (this.dirty.size > 0) {
      requestAnimationFrame(() => this.flush());
    } else {
      this.running = false;
    }
  }
}

class Semaphore {
  constructor(limit) {
    this.limit = limit;
    this.active = 0;
    this.queue = [];
  }

  acquire() {
    return new Promise(resolve => {
      if (this.active < this.limit) {
        this.active++;
        resolve();
      } else {
        this.queue.push(resolve);
      }
    });
  }

  release() {
    this.active--;

    const next = this.queue.shift();

    if (next) {
      this.active++;
      next();
    }
  }
}

const globalSemaphore = new Semaphore(8);

async function runRequest(fn) {
  await globalSemaphore.acquire();

  try {
    return await fn();
  } finally {
    globalSemaphore.release();
  }
}

JavaScript
function mapReasoning({
  protocol,
  requested,
  capabilities
}) {
  const supported =
    capabilities?.reasoningLevels || [];

  if (requested === "auto") {
    return { mode: "auto" };
  }

  const order = [
    "minimal",
    "low",
    "medium",
    "high",
    "max"
  ];

  // 从请求等级开始，向下寻找模型真正支持的等级
  let index = order.indexOf(requested);

  while (index >= 0) {
    const level = order[index];

    if (supported.includes(level)) {
      switch (protocol) {
        case "openai-responses":
          return {
            reasoning: {
              effort:
                level === "minimal"
                  ? "none"
                  : level
            }
          };

        case "openai-chat":
          return {
            reasoning_effort:
              level === "minimal"
                ? "none"
                : level
          };

        case "anthropic-messages":
          return {
            output_config: {
              effort:
                level === "minimal"
                  ? "low"
                  : level
            }
          };

        case "gemini-native":
          return {
            thinking_level:
              level === "minimal"
                ? "minimal"
                : level
          };

        case "ollama":
          return {
            think:
              level === "minimal"
                ? false
                : level
          };
      }
    }

    index--;
  }

  return {
    unsupported: true
  };
}
