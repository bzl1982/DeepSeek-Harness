## 💡 核心办法：换掉 executeJavaScript，改用 Electron 的 insertText 和 sendInputEvent
这是最推荐的“大招”。它不需要你苦苦寻找 React/Vue 的内部状态、不需要考虑按钮叫什么、更不怕 Shadow DOM。它直接通过 Chromium 内核向网页发送原生系统级事件，网页框架会百分之百认为这是“用户自己敲键盘输入的”。
## 1. 主进程/渲染进程控制逻辑（核心代码）
当用户在底部输入框点击发送时，循环对 9 个 webview 执行以下序列：

async function broadcastToWebview(webviewElement, text) {
  // 1. 让当前的 webview 获得焦点
  webviewElement.focus();

  // 2. 让网页里的输入框聚焦（这一步只需要一个极其泛化的模糊选择器）
  await webviewElement.executeJavaScript(`
    (() => {
      // 寻找最像输入框的元素
      const input = document.querySelector('textarea, div[contenteditable="true"], [role="textbox"]');
      if (input) {
        input.focus();
        // 顺便全选旧内容（防止第二轮失效时文字叠加）
        if (input.tagName === 'TEXTAREA' || input.tagName === 'INPUT') {
          input.select();
        } else {
          const range = document.createRange();
          range.selectNodeContents(input);
          const sel = window.getSelection();
          sel.removeAllRanges();
          sel.addRange(range);
        }
      }
    })()
  `);

  // 给浏览器 50ms 响应聚焦
  await new Promise(r => setTimeout(r, 50));

  // 3. 极其关键：利用 Electron 原生能力把文本“硬塞”进去
  // 这会直接触发 React/Vue 的底层监听，输入框一定会亮起，发送按钮一定会变激活状态！
  webviewElement.insertText(text);

  // 给框架 100ms 时间更新它的内部 State
  await new Promise(r => setTimeout(r, 100));

  // 4. 模拟按下并抬起 Enter 键发送
  // 绝大多数 AI 网页在输入框聚焦且有字时，敲 Enter 就能直接发送
  webviewElement.sendInputEvent({ type: 'keyDown', keyCode: 'Enter' });
  webviewElement.sendInputEvent({ type: 'keyUp', keyCode: 'Enter' });
}

## 🛠️ 针对顽固分子的特殊补丁（解决 Kimi、DeepSeek、Gemini）
如果使用上面的通用方法后，有个别网站（如豆包、通义）按 Enter 没有发送，而是换行了，或者没点到按钮，可以在 preload.js 里加入以下精准防坑补丁：
## 补丁 A：利用 document.execCommand 强行破除 React/Vue 状态锁
对于文字留在了输入框，但发送按钮没亮（没触发状态更新）的网站（如豆包、通义千问），在注入文字时改用这个：

// 在 webview 的控制脚本中function injectTextSuper(el, text) {
  el.focus();
  // 选定所有并用命令替换，100% 触发前端框架的 Data Binding
  document.execCommand('selectAll', false, null);
  document.execCommand('insertText', false, text);
  // 额外补一个冒泡事件
  el.dispatchEvent(new Event('input', { bubbles: true, cancelable: true }));
}

## 补丁 B：穿透 Gemini 的 Shadow DOM 寻找器
Gemini 找不到输入框是因为它被包裹在 Shadow Root 里面，普通的 querySelector 穿不过去。用这个函数可以实现全网页深层穿透：

function findAbsoluteInput(root = document) {
  // 1. 找当前层级
  const input = root.querySelector('textarea:not([type="hidden"]), div[contenteditable="true"], [role="textbox"]');
  if (input && input.offsetParent !== null) return input; // 确保可见

  // 2. 穿透 Shadow DOM 递归查找
  const all = root.querySelectorAll('*');
  for (const el of all) {
    if (el.shadowRoot) {
      const found = findAbsoluteInput(el.shadowRoot);
      if (found) return found;
    }
  }
  return null;
}
