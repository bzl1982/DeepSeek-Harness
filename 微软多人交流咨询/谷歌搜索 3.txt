## 1.1 客户端 Drop 拦截与真实路径提取（修改 meeting.html）
利用 Electron 环境下 File 对象特有的 path 属性（原生 Web 浏览器中不可见，但 Electron 中可直接拿到绝对路径），彻底重构文件解析：

// 目标文件：desktop/agent/meeting.html - 修改 handleFiles 函数function handleFiles(files) {
  const container = document.getElementById('attachment-list');
  
  for (let i = 0; i < files.length; i++) {
    const file = files[i];
    // ★ 关键：提取 Electron 注入的本地绝对路径（地基补全）
    const localPath = file.path || ''; 
    const ext = file.name.split('.').pop().toLowerCase();
    
    const fileItem = {
      id: 'att_' + Date.now() + '_' + Math.random().toString(36).substr(2, 5),
      name: file.name,
      size: file.size,
      localPath: localPath, // 注入物理路径，供 AttachmentBus 消费
      ext: ext,
      status: 'pending'
    };

    // 区分文件类型，提取预览或文本
    if (['txt', 'md', 'markdown', 'json', 'csv', 'log'].includes(ext)) {
      const reader = new FileReader();
      reader.onload = function(e) {
        fileItem.text = e.target.result;
        attachedFiles.push(fileItem);
        renderAttachmentCard(fileItem);
      };
      reader.readAsText(file);
    } else {
      // PDF、图片、Docx 等非纯文本格式
      // 不再使用无意义的 [PDF文件: xxx] 文本占位符污染发送框
      fileItem.text = `[FileAttachment: ${file.name}]`; 
      if (file.type.startsWith('image/')) {
        fileItem.isImage = true;
        fileItem.dataUrl = URL.createObjectURL(file); // 仅做前端 UI 预览预览，不参与文本拼接
      }
      attachedFiles.push(fileItem);
      renderAttachmentCard(fileItem);
    }
  }
}

## 🛡️ 三、 防御性完成检测（解决“抓空”与“跨轮闩锁”Bug）
针对谷歌 2 的代码缺陷（AI 尚未开始吐字、页面处于初始静止状态时，文本无增长信号误置为真，导致直接提前结束并抓取到空串） [2]，在测试台中实施“必须首见文本（RequireContent）”和“信号状态激活器（Arm 机制）”：

// meeting-lab/core/completion.js - 完善防守逻辑
class SecureCompletionDetector {
  constructor(webContents, selectors) {
    this.webContents = webContents;
    this.selectors = selectors;
    this.armed = false; // 每轮初始化状态锁
  }

  /**
   * 每一轮发送问题前，必须手动重置并激活状态锁，杜绝跨轮信号污染（解决跨轮闩锁） [2]
   */
  arm() {
    this.armed = true;
    this.hasSeenContent = false; // 必须首先看到内容产生
  }

  async waitForResponse(timeoutMs = 45000) {
    if (!this.armed) {
      throw new Error("[Detector] 未执行 arm() 激活，严禁进入完成检测周期");
    }

    let lastText = "";
    let silenceTicks = 0;

    return new Promise((resolve) => {
      const timeoutTimer = setTimeout(() => {
        clearInterval(loop);
        this.armed = false;
        resolve({ status: 'timeout', data: '【闭麦提示：响应超时】' });
      }, timeoutMs);

      const loop = setInterval(async () => {
        try {
          const [hasStop, hasSend, currentText] = await this.webContents.executeJavaScript(`
            (() => {
              const stop = !!document.querySelector('${this.selectors.stopBtn}');
              const send = !!document.querySelector('${this.selectors.sendBtn}');
              
              // 穿透 Shadow DOM 支持（Hy4 范式：通过 preload 注册获取） [2]
              let text = '';
              const el = document.querySelector('${this.selectors.messageText}');
              if (el) text = el.innerText || el.textContent || '';
              return [stop, send, text];
            })()
          `);

          // 核心防御点一：判断 AI 是否真正开始吐字 [2]
          if (!this.hasSeenContent && currentText.trim().length > 0) {
            this.hasSeenContent = true; // AI 开始输出，RequireContent 门槛跨越 [2]
          }

          // 如果 AI 还没开始吐字，即使页面完全静止，也绝对不允许判定为结束 [2]
          if (!this.hasSeenContent) {
            return; // 继续等待生成开始
          }

          // 核心防御点二：文本静止计数
          if (currentText === lastText) {
            silenceTicks++;
          } else {
            silenceTicks = 0;
            lastText = currentText;
          }

          // 满足联合判据：AI 吐过字 + 停止按钮消失 + 发送恢复 + 文本连续 5 次采样（1.5秒）无变化 [2]
          if (!hasStop && hasSend && silenceTicks >= 5) {
            clearInterval(loop);
            clearTimeout(timeoutTimer);
            this.armed = false;
            resolve({ status: 'completed', data: currentText.trim() });
          }
        } catch (err) {
          // 忽略执行偶发异常
        }
      }, 300);
    });
  }
}

## 🛡️ 五、 桌面端专属：同一设备指纹多账号防风控规约（盲点 B 突破）
9 个账号在同一套 Electron 宿主、同一 IP 下高频并发，极易被反爬判别为“账号农场” [2]。必须在 meeting-lab/core/orchestrator.js 的并发执行器中接入流量整形与安全闭麦熔断：

// meeting-lab/core/orchestrator.js - 引入风控整形限制
class SecureMeetingOrchestrator {
  constructor(agents, concurrencyLimit = 3) {
    this.agents = agents;
    this.concurrencyLimit = concurrencyLimit; // 限制同时处于生成状态的浏览器上限 [2]
  }

  async executeBroadcast(promptInput) {
    console.log(`[Security] 启动防风控错峰广播流量整形，并发限制: ${this.concurrencyLimit}`);
    
    const results = {};
    let activeGenerations = 0;
    
    // 使用队列池控制并发上限，阻止 9 个网页在一瞬间同时点击发送
    const pool = this.agents.map((agent, index) => async () => {
      // 错峰延迟：依据各自的格子索引，人为制造 400ms - 2000ms 的随机拟人打字/点击间隔 [2]
      const staggeredDelay = index * 400 + Math.random() * 300;
      await new Promise(r => setTimeout(r, staggeredDelay));

      try {
        agent.updateState('pending');
        await agent.adapter.sendText(promptInput);
        
        // 阻塞等待响应
        const res = await agent.detector.waitForResponse(45000);
        
        // 关键防御点：如果检测到页面弹出了滑动验证码
        if (res.data.includes("请完成安全验证") || res.data.includes("captcha")) {
          throw new Error("CAPTCHA_TRIGGERED");
        }

        results[agent.id] = res;
      } catch (err) {
        if (err.message === "CAPTCHA_TRIGGERED") {
          console.error(`[CRITICAL RISK] AI【${agent.name}】触发滑动验证码！`);
          // 熔断规约：一旦触碰验证码，严禁自动重试风暴，立即原地闭麦，交由人工处理 [2]
          agent.updateState('blocked_by_captcha'); 
          return { status: 'failed', content: '【⚠️ 系统提示：该AI当前触发风控验证码，已自动为您就地闭麦。请在宫格内手动完成滑块验证后右键唤醒】' };
        }
        
        agent.updateState('failed');
        return { status: 'failed', content: `【报错闭麦】: ${err.message}` };
      }
    });

    // 顺序与并发调度器控制
    const chunkedTasks = [];
    for (let i = 0; i < pool.length; i += this.concurrencyLimit) {
      const chunk = pool.slice(i, i + this.concurrencyLimit).map(task => task());
      await Promise.allSettled(chunk); // 分批扇出，减缓宿主指纹的短时流量洪峰 [2]
    }

    return results;
  }
}
