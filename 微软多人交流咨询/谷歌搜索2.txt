## 🛠️ 核心突破一：联合信号完成检测器（completion.js）
网页版 AI 最致命的 Bug 就是“还没吐完，系统误以为说完了”。
我们采用多信号联合判定（CDP 网络流结束 + DOM 状态 + 文本静止检测） [1] 的漏斗模型，并加入“闭麦降级”超时机制。

// meeting-lab/core/completion.jsconst EventEmitter = require('events');
class CompletionDetector extends EventEmitter {
  /**
   * @param {Object} webContents - Electron 的 webContents 实例
   * @param {Object} selectors - 当前 AI 的 DOM 契约选择器
   */
  constructor(webContents, selectors) {
    super();
    this.webContents = webContents;
    this.selectors = selectors; // 包含 stopBtn, sendBtn, messageText 等
  }

  /**
   * 启动联合检测核心方法
   * @param {number} timeoutMs - 超时时间（默认 45 秒）
   */
  async waitForResponse(timeoutMs = 45000) {
    let isFinished = false;
    let textSilenceTimer = null;
    let lastTextLength = 0;
    let networkActive = true;

    // 1. 开启 CDP (Chrome DevTools Protocol) 监听网络流 (信号 D)
    const client = this.webContents.debugger;
    if (!client.isAttached()) {
      client.attach();
    }
    client.sendCommand('Network.enable');

    // 监听网络请求结束
    const onLoadingFinished = () => { networkActive = false; };
    client.on('Network.loadingFinished', onLoadingFinished);
    client.on('Network.loadingFailed', onLoadingFinished);

    return new Promise((resolve, reject) => {
      // 安全超时兜底：超时直接“闭麦”标记为超时，让会议继续 [1]
      const timeoutTimer = setTimeout(() => {
        cleanup();
        resolve({ status: 'timeout', data: '【系统提示：该AI由于网络/渲染原因响应超时，暂时闭麦。】' }); [1]
      }, timeoutMs);

      const cleanup = () => {
        clearTimeout(timeoutTimer);
        clearInterval(checkLoop);
        if (textSilenceTimer) clearTimeout(textSilenceTimer);
        client.off('Network.loadingFinished', onLoadingFinished);
        client.off('Network.loadingFailed', onLoadingFinished);
      };

      // 2. 主检查循环 (每 300ms 采样一次 DOM 和网络状态)
      const checkLoop = setInterval(async () => {
        try {
          // 信号 A & B：检查“停止”按钮是否存在，“发送”按钮是否恢复可用
          const [hasStopBtn, hasSendBtn, currentText] = await this.webContents.executeJavaScript(`
            (() => {
              const stop = !!document.querySelector('${this.selectors.stopBtn}');
              const send = !!document.querySelector('${this.selectors.sendBtn}');
              
              // 穿透 Shadow DOM (处理豆包/通义等特殊站点) [1]
              const getDeepText = () => {
                const nodes = document.querySelectorAll('${this.selectors.messageText}');
                if(!nodes.length) return '';
                const lastNode = nodes[nodes.length - 1];
                return lastNode.innerText || lastNode.textContent || '';
              };
              
              return [stop, send, getDeepText()];
            })()
          `);

          // 信号 C：文本静止检测 (连续 1500ms 长度无变化) [1]
          if (currentText.length > 0 && currentText.length === lastTextLength) {
            if (!textSilenceTimer) {
              textSilenceTimer = setTimeout(() => {
                // 文本静止信号触发
                evaluateWeights(true, hasStopBtn, hasSendBtn, currentText);
              }, 1500);
            }
          } else {
            // 文本还在增长，重置静止计时器
            if (textSilenceTimer) {
              clearTimeout(textSilenceTimer);
              textSilenceTimer = null;
            }
            lastTextLength = currentText.length;
          }

          // 兜底状态评估
          evaluateWeights(false, hasStopBtn, hasSendBtn, currentText);

        } catch (err) {
          // 忽略执行期间的偶发 DOM 销毁错误
        }
      }, 300);

      // 决策矩阵（多信号联合评分） [1]
      function evaluateWeights(textIsSilent, hasStop, hasSend, text) {
        let score = 0;
        if (!hasStop) score += 1;     // 信号 A: 停止按钮消失 [1]
        if (hasSend) score += 1;      // 信号 B: 发送按钮恢复 [1]
        if (textIsSilent) score += 1; // 信号 C: 文本无增长 [1]
        if (!networkActive) score += 1; // 信号 D: 网络无大流量推流 [1]

        // 满足 3 个以上强信号，或者按钮完全就绪且有文本 [1]
        if (score >= 3 || (!hasStop && hasSend && text.length > 0)) {
          cleanup();
          resolve({ status: 'completed', data: text.trim() });
        }
      }
    });
  }
}

module.exports = CompletionDetector;

## 🔗 核心突破二：分布式附件门闩总线（attachment-bus.js）
解决“9 个人偶尔收不到文件”的核心：废除固定的 sleep(3000)，建立严格的 fan-out / fan-in 握手 ACK 机制。

// meeting-lab/core/attachment-bus.jsconst fs = require('fs');const path = require('path');
class AttachmentBus {
  constructor(meetingCore) {
    this.meetingCore = meetingCore; // 注入核心，用于调度各 WebAI 实例的 CDP
  }

  /**
   * 统一接管拖拽/上传文件，落盘并分发 [1]
   * @param {string} filePath - 本地物理路径
   */
  async registerAndDistribute(filePath) {
    if (!fs.existsSync(filePath)) {
      throw new Error(`文件不存在: ${filePath}`);
    }

    const stats = fs.statSync(filePath);
    const attachment = {
      attachmentId: `att_${Date.now()}_${Math.random().toString(36).substr(2, 5)}`, [1]
      name: path.basename(filePath),
      size: stats.size,
      localPath: filePath
    };

    // 准备分发到当前参会的全部活跃 Agents
    const activeAgents = this.meetingCore.getActiveAgents(); // 返回当前格子里的 AI 列表
    
    console.log(`[AttachmentBus] 开始向 ${activeAgents.length} 个 AI 分发文件握手...`);

    // 并发扇出：让所有 Webview 并发通过 CDP 塞入文件 [1]
    const uploadPromises = activeAgents.map(async (agent) => {
      try {
        // 1. 调用底层的 CDP setFileInputFiles 注入文件（免去模拟点击） [1]
        await agent.adapter.injectFileViaCDP(attachment.localPath);

        // 2. 核心：开启 ACK 门闩，等待该网页端生成对应的附件卡片 DOM (带 15s 超时) [1]
        const ackSuccess = await agent.adapter.waitForUploadACK(15000);
        
        if (!ackSuccess) {
          throw new Error('网页端接收文件超时，未出现附件卡片');
        }

        return { agentId: agent.id, success: true };
      } catch (err) {
        console.error(`[AttachmentBus] AI【${agent.name}】文件接收失败:`, err.message);
        // 单点失败降级：转换为文本摘要标记，防止卡死整场会议 [1]
        await agent.adapter.injectTextFallback(`[本地附件解析失败提示: 上传了文件 ${attachment.name}]`);
        return { agentId: agent.id, success: false, error: err.message };
      }
    });

    // 门闩同步点：所有 AI 必须回报状态 [1]
    const results = await Promise.all(uploadPromises);
    const failures = results.filter(r => !r.success);

    if (failures.length > 0) {
      console.warn(`[AttachmentBus] 警告: 有 ${failures.length} 个 AI 未成功解析文件，已执行优雅降级。`); [1]
    } else {
      console.log(`[AttachmentBus] 门闩释放！所有 AI 文件同步就绪，允许放行发送指令。`); [1]
    }

    return attachment;
  }
}

module.exports = AttachmentBus;

## 🧠 核心突破三：纯 Node 编排内核与单点隔离（orchestrator.js）
不依赖微软框架运行时，直接使用 Promise.allSettled 构建支持广播模式、轮询模式，具备单点挂起、容错自愈能力的会议大脑。

// meeting-lab/core/orchestrator.js
class MeetingOrchestrator {
  constructor(agents, speakerStrategy = 'broadcast') {
    this.agents = agents; // 实例化的 WebAI Agent 数组
    this.speakerStrategy = speakerStrategy; // broadcast | round_robin [1]
    this.meetingHistory = []; // 会议消息统一存储，独立于网页 [1]
    this.currentRound = 0;
  }

  /**
   * 投递用户问题或上一轮结论进行激发
   * @param {string} promptInput - 输入内容
   */
  async dispatchContext(promptInput) {
    this.currentRound++;
    console.log(`\n--- [通辽会议] 进入第 ${this.currentRound} 轮辩论 ---`);

    const turnMessage = {
      round: this.currentRound,
      timestamp: Date.now(),
      input: promptInput,
      responses: {}
    };

    if (this.speakerStrategy === 'broadcast') {
      // 🚀 场景 A：一发多收（广播模式）
      // 使用 allSettled 确保单点 AI 卡死或报验证码时，不影响其他 8 个正常输出 [1]
      const tasks = this.agents.map(async (agent) => {
        return this.executeAgentTurn(agent, promptInput);
      });

      const results = await Promise.allSettled(tasks);
      
      results.forEach((res, index) => {
        const agent = this.agents[index];
        if (res.status === 'fulfilled') {
          turnMessage.responses[agent.id] = res.value;
        } else {
          turnMessage.responses[agent.id] = {
            status: 'failed',
            content: `【系统提示：当前AI发生致命异常：${res.reason}】` [1]
          };
        }
      });

    } else if (this.speakerStrategy === 'round_robin') {
      // 🔄 场景 B：轮询/接力会议模式 [1]
      let currentInput = promptInput;
      for (const agent of this.agents) {
        // 在 Prompt 中动态注入“前一个 AI 说了什么”，实现交叉辩论 [1]
        const promptWithContext = `会议历史记录：\n${currentInput}\n\n请针对以上内容发表你的独立见解。`; [1]
        const result = await this.executeAgentTurn(agent, promptWithContext);
        turnMessage.responses[agent.id] = result;
        
        // 滚动更新上下文
        if (result.status === 'completed') {
          currentInput += `\n【${agent.name}】说: ${result.content}`;
        }
      }
    }

    this.meetingHistory.push(turnMessage);
    return turnMessage;
  }

  /**
   * 单个 Agent 的执行单元状态机 [1]
   */
  async executeAgentTurn(agent, prompt) {
    console.log(`[Orchestrator] 驱动 【${agent.name}】 发言...`);
    
    try {
      // 1. 设置 AI 状态机为 pending [1]
      agent.updateState('pending'); 

      // 2. 网页端物理发送
      await agent.adapter.sendText(prompt);
      agent.updateState('streaming');

      // 3. 阻塞调用核心突破一的“联合完成检测器”
      const reply = await agent.detector.waitForResponse(45000);

      // 4. 更新最终状态
      agent.updateState(reply.status); // completed 或 timeout [1]
      
      return {
        status: reply.status,
        content: reply.data
      };

    } catch (error) {
      agent.updateState('failed');
      return {
        status: 'failed',
        content: `【报错闭麦】: ${error.message}`
      };
    }
  }
}

module.exports = MeetingOrchestrator;

## 🌍 核心突破四：WebAI 契约抽象层扩展（以已登录的 DeepSeek 为例）
为了完美复用您 Electron 目录中躺着的 Partitions/ 登录态 [1]，Adapter 层必须补齐 CDP 文件穿透注入和 ACK 门闩检测：

// meeting-lab/adapters/deepseek-web.js
class DeepSeekWebAdapter {
  constructor(webviewInstance) {
    this.webview = webviewInstance; 
  }

  /**
   * 关键补齐：利用 CDP 绕过物理点击，直接向网页文件框强制塞入文件 [1]
   */
  async injectFileViaCDP(absoluteFilePath) {
    const client = this.webview.getWebContents().debugger;
    if (!client.isAttached()) client.attach();

    // 1. 获取目标网页的文件输入 DOM 节点句柄
    const { root } = await client.sendCommand('DOM.getDocument');
    const { nodeId } = await client.sendCommand('DOM.querySelector', {
      nodeId: root.nodeId,
      selector: 'input[type="file"]' // 对应 DeepSeek 或通用站点的上传控件
    });

    // 2. 强行塞入本地路径，直接触发前端上传逻辑，不弹窗 [1]
    await client.sendCommand('DOM.setFileInputFiles', {
      nodeId: nodeId,
      files: [absoluteFilePath]
    });
  }

  /**
   * 关键补齐：检测网页端是否已经“收齐”并解析完了附件（ACK） [1]
   */
  async waitForUploadACK(timeoutMs = 15000) {
    const startTime = Date.now();
    while (Date.now() - startTime < timeoutMs) {
      // 检测 DOM 中是否出现了解析成功的标记（如“删除附件”按钮可用、或进度条到100%） [1]
      const isUploaded = await this.webview.executeJavaScript(`
        (() => {
          // 这里的 Selector 依据各家而定，DeepSeek 出现 class 包含 "ds-icon" 且在卡片内即可
          const fileCard = document.querySelector('.ds-upload-list-item');
          if (!fileCard) return false;
          // 检查是否还在 loading 状态
          const isContextLoading = fileCard.querySelector('.loading-spinner, .progress-active');
          return !isContextLoading; 
        })()
      `);
      
      if (isUploaded) return true;
      await new Promise(r => setTimeout(r, 400)); // 轮询阻尼
    }
    return false; // 超时未收到 ACK
  }

  async sendText(text) {
    // 驱动已有的输入框物理发送
    await this.webview.executeJavaScript(`
      (() => {
        const textarea = document.querySelector('textarea');
        textarea.value = \`\${text.replace(/`/g, '\\`').replace(/\$/g, '\\$')}\`;
        textarea.dispatchEvent(new Event('input', { bubbles: true }));
        setTimeout(() => {
          const btn = document.querySelector('.ds-send-button');
          if(btn) btn.click();
        }, 100);
      })()
    `);
  }
}

module.exports = DeepSeekWebAdapter;
